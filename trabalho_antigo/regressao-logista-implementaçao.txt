library(ISLR)
library(gridExtra)
library(pROC)

set.seed(42)

db <- ISLR::Default

view(db)

g1 <- db |> 
	ggplot(aes(x = balance, y = income, fill = default)) + geom_point() + theme_bw()  

g2 <- db |> 
	ggplot(aes(x = default, y = income, fill = default)) + geom_boxplot() + theme_bw() 

g3 <- db |> 
	ggplot(aes(x = default, y = balance, fill = default)) + geom_boxplot() + theme_bw() 

g4 <-  db |> 
	ggplot(aes(x = student, y = balance, fill = student)) + geom_boxplot() + theme_bw() 

table1 <- table(x =  db$student, y = db$default)

prop.table(table1, margin = 1)

g5 <- db |> 
  mutate(default_num = ifelse(default == "Yes", 1, 0),
    balance_bin = cut(balance, breaks = 20)) |> 
  group_by(student, balance_bin) |> 
  summarise(taxa_default = mean(default_num),
    balance_medio = mean(balance, na.rm = TRUE)) |> 
  ggplot(aes(x = balance_medio, y = taxa_default, color = student)) +
  geom_line(linewidth = 1) +
  geom_point() +
  theme_bw() +
  labs(x = "Balance (Médio por faixa)",
    y = "Taxa Empírica de Default",
    color = "Student") + 
	xlim(0, 2000) + 
	geom_hline(yintercept = 0.043, colour = "#00BFC4", linetype = "dashed") + 
	geom_hline(yintercept = 0.029, colour = "#F8766D", linetype = "dashed")

gridExtra::grid.arrange(g5, g4, nrow = 1)

idx <- sample(1:nrow(db), replace = TRUE, size = round(0.5*nrow(db)))
training <- db[idx, ]
test <- db[-idx, ]

model <- glm(default ~ balance + income + student,
	family = "binomial", data = training)

summary(model)

pi_hat <- predict(model, newdata = test, type = "response")

threshold <- 0.5

phi_hat <- factor(ifelse(pi_hat >= threshold, "Yes", "No"))

table(Predicted = phi_hat, Observed = test$default)

par(mfrow= c(1,1))
roc_default <- roc(test$default, pi_hat)

plot(roc_default, col = "blue", grid = TRUE, 
	legacy.axes = TRUE, asp = FALSE, las = 1,
	print.thres.pattern = "%.3f",
	print.thres = "best")

coords(roc_default, "best", best.method = "youden")
coords(roc_default, "best", best.method = "closest.topleft")
